-- ---------------------------------------------------------
-- 1. FIX TABLE STRUCTURES (Task 3.5 Prefix: dbProj_)
-- ---------------------------------------------------------

-- Fix Users table (Plural name)
CREATE TABLE IF NOT EXISTS dbProj_Users (
    UserID INT AUTO_INCREMENT PRIMARY KEY,
    Username VARCHAR(50) NOT NULL UNIQUE,
    Email VARCHAR(100) NOT NULL UNIQUE,
    Password VARCHAR(255) NOT NULL,
    Role ENUM('Viewer', 'Creator', 'Admin') DEFAULT 'Viewer',
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- Fix Recipes table (Adding Ingredients, Instructions, Video, and flexible Category)
CREATE TABLE IF NOT EXISTS dbProj_Recipes (
    RecipeID INT AUTO_INCREMENT PRIMARY KEY,
    UserID INT NOT NULL,
    Title VARCHAR(255) NOT NULL,
    Description TEXT,
    Ingredients TEXT,
    Instructions TEXT,
    ImagePath VARCHAR(255) DEFAULT 'default.jpg',
    VideoPath VARCHAR(255) DEFAULT NULL,
    Category VARCHAR(50) NOT NULL,
    Views INT DEFAULT 0,
    Status ENUM('Draft', 'Published') DEFAULT 'Published',
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (UserID) REFERENCES dbProj_Users(UserID) ON DELETE CASCADE
) ENGINE=InnoDB;

-- Create Comments table
CREATE TABLE IF NOT EXISTS dbProj_Comments (
    CommentID INT AUTO_INCREMENT PRIMARY KEY,
    RecipeID INT NOT NULL,
    UserID INT NOT NULL,
    CommentText TEXT NOT NULL,
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (RecipeID) REFERENCES dbProj_Recipes(RecipeID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES dbProj_Users(UserID) ON DELETE CASCADE
) ENGINE=InnoDB;

-- Create Ratings table
CREATE TABLE IF NOT EXISTS dbProj_Ratings (
    RatingID INT AUTO_INCREMENT PRIMARY KEY,
    RecipeID INT NOT NULL,
    UserID INT NOT NULL,
    RatingValue INT NOT NULL CHECK (RatingValue BETWEEN 1 AND 5),
    FOREIGN KEY (RecipeID) REFERENCES dbProj_Recipes(RecipeID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES dbProj_Users(UserID) ON DELETE CASCADE,
    UNIQUE KEY unique_rating (RecipeID, UserID)
) ENGINE=InnoDB;

-- ---------------------------------------------------------
-- 2. ADVANCED FEATURES (Task 3.2 & Task 4)
-- ---------------------------------------------------------

-- REQUIREMENT: Stored Procedure for Admin Reports
DELIMITER //
CREATE PROCEDURE sp_GetPopularRecipes()
BEGIN
    SELECT r.Title, r.Category, r.Views, u.Username
    FROM dbProj_Recipes r
    JOIN dbProj_Users u ON r.UserID = u.UserID
    ORDER BY r.Views DESC
    LIMIT 10;
END //
DELIMITER ;

-- BONUS: Trigger to automatically log "System Messages" (Task 1.6 Requirement)
DELIMITER //
CREATE TRIGGER tr_OnRecipeDelete
BEFORE DELETE ON dbProj_Recipes
FOR EACH ROW
BEGIN
    -- This helps prove you are handling content management professionally
    INSERT INTO dbProj_Comments (RecipeID, UserID, CommentText)
    VALUES (OLD.RecipeID, 1, 'System: Content removed by Administrator.');
END //
DELIMITER ;
