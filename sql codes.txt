-- 1. Create Users Table
CREATE TABLE IF NOT EXISTS dbProj_User (
    UserID INT AUTO_INCREMENT PRIMARY KEY,
    Username VARCHAR(50) NOT NULL UNIQUE,
    Email VARCHAR(100) NOT NULL UNIQUE,
    Password VARCHAR(255) NOT NULL,
    Role ENUM('Viewer', 'Creator', 'Admin') DEFAULT 'Viewer',
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 2. Create Recipes Table
CREATE TABLE IF NOT EXISTS dbProj_Recipes (
    RecipeID INT AUTO_INCREMENT PRIMARY KEY,
    UserID INT NOT NULL,
    Title VARCHAR(255) NOT NULL,
    Description TEXT,
    Ingredients TEXT,
    Instructions TEXT,
    ImagePath VARCHAR(255) DEFAULT 'default.jpg',
    VideoPath VARCHAR(255),
    Category VARCHAR(50),
    Views INT DEFAULT 0,
    RatingSum INT DEFAULT 0,
    RatingCount INT DEFAULT 0,
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    Status ENUM('Draft', 'Published') DEFAULT 'Published',
    FOREIGN KEY (UserID) REFERENCES dbProj_User(UserID) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 3. Create Comments Table
CREATE TABLE IF NOT EXISTS dbProj_Comments (
    CommentID INT AUTO_INCREMENT PRIMARY KEY,
    RecipeID INT NOT NULL,
    UserID INT NOT NULL,
    CommentText TEXT NOT NULL,
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (RecipeID) REFERENCES dbProj_Recipes(RecipeID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES dbProj_User(UserID) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 4. Create Ratings Table (Needed for functionality)
CREATE TABLE IF NOT EXISTS dbProj_Ratings (
    RatingID INT AUTO_INCREMENT PRIMARY KEY,
    RecipeID INT NOT NULL,
    UserID INT NOT NULL,
    RatingValue INT NOT NULL CHECK (RatingValue BETWEEN 1 AND 5),
    FOREIGN KEY (RecipeID) REFERENCES dbProj_Recipes(RecipeID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES dbProj_User(UserID) ON DELETE CASCADE,
    UNIQUE KEY unique_rating (RecipeID, UserID)
) ENGINE=InnoDB;

-- 5. Stored Procedure for Search (Requirement: At least one stored procedure)
DELIMITER //
CREATE PROCEDURE sp_SearchRecipes(IN searchTerm VARCHAR(255))
BEGIN
    SELECT r.RecipeID, r.Title, r.ImagePath, r.Description, u.Username, AVG(rt.RatingValue) as AvgRating
    FROM dbProj_Recipes r
    JOIN dbProj_User u ON r.UserID = u.UserID
    LEFT JOIN dbProj_Ratings rt ON r.RecipeID = rt.RecipeID
    WHERE r.Status = 'Published' 
    AND (r.Title LIKE CONCAT('%', searchTerm, '%') OR r.Ingredients LIKE CONCAT('%', searchTerm, '%'))
    GROUP BY r.RecipeID
    ORDER BY r.CreatedAt DESC;
END //
DELIMITER ;
